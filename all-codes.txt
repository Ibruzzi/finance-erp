# =============================================================================
# Script: concatenate_code.sh
# Purpose: Concatenate all code files (including .manifest) from a project
#          directory into a single .txt file, excluding specified folders
#          and specific files.
# =============================================================================

# --- Configuration -----------------------------------------------------------
SOURCE_DIR="/home/iburu/finance-erp"

# --- Write output ------------------------------------------------------------
: > "$OUTPUT_FILE"

# Header
{
    echo "================================================================"
    echo "  CODE CONCATENATION DUMP"
    echo "  Source:  $SOURCE_DIR"
    echo "  Created: $(date)"
    echo "  Files:   ${#FILES[@]}"
    echo "================================================================"
    echo
} >> "$OUTPUT_FILE"

# Concatenate each file
count=0
for file in "${FILES[@]}"; do
    rel_path="${file#$SOURCE_DIR/}"
    count=$((count + 1))

    {
        echo
        echo "================================================================"
        echo "FILE: $rel_path"
        echo "================================================================"
        cat "$file"
        echo
        echo "--- END OF FILE: $rel_path ---"
    } >> "$OUTPUT_FILE"
done

# --- Report ------------------------------------------------------------------
echo "Done. Concatenated $count files."
echo "Output saved to: $OUTPUT_FILE"
echo "Size: $(du -h "$OUTPUT_FILE" | cut -f1)"

// public/sw.js

const CACHE_NAME = 'finance-erp-v2';

self.addEventListener('install', () => {
  self.skipWaiting();
});

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then((cacheNames) => {
      return Promise.all(
        cacheNames.map((cacheName) => {
          if (cacheName !== CACHE_NAME) {
            return caches.delete(cacheName);
          }
        })
      );
    })
  );
  self.clients.claim();
});

self.addEventListener('fetch', (event) => {
  if (event.request.method !== 'GET') return;

  // Never cache Supabase API or auth traffic:
  // - API responses are user-scoped and would leak between sessions.
  // - Cached API data would short-circuit fresh pulls.
  try {
    const url = new URL(event.request.url);
    if (url.pathname.startsWith('/rest/') || url.pathname.startsWith('/auth/')) {
      return;
    }
  } catch {
    // If URL parsing fails, fall through to normal handling.
  }

  event.respondWith(
    fetch(event.request)
      .then((response) => {
        const responseClone = response.clone();
        caches.open(CACHE_NAME).then((cache) => {
          cache.put(event.request, responseClone);
        });
        return response;
      })
      .catch(() => {
        return caches.match(event.request);
      })
  );
});

import { useEffect, useRef, useState } from 'react';
import { Capacitor } from '@capacitor/core';
import { App as CapacitorApp } from '@capacitor/app';
import { useStore } from './store/useStore';

import { supabase } from './lib/supabaseClient';
import { syncAll } from './lib/sync';
import { AuthGate } from './components/AuthGate';
import { isCloudSyncOn } from './lib/cloudMode';
import { SyncStatus } from './components/SyncStatus';
import { LoadingScreen } from './components/LoadingScreen';

import { MonthPicker } from './components/MonthPicker';
import { AppTabs } from './components/AppTabs';
import { ZakatCard } from './components/ZakatCard';
import { TotalDebtCard } from './components/TotalDebtCard';
import { CategoryTotals } from './components/CategoryTotals';

  { key: 'dashboard', label: 'Dashboard' },
  { key: 'insights', label: 'Insights' },
  { key: 'summary', label: 'Summary' },
  { key: 'range', label: 'Range' },

function MainApp() {
  const {
    loadData, isLoading, settings, getBucketBalances, getCategoryTotals,
    calculateZakat, addTransaction, updateTransaction, deleteTransaction,
    getMonthTransactions, bucketMap, getBucketFromMap, getSubcategoriesFor,
    getCategories, updateSetting, loans, getLoanPaid, getTotalDebtPayable, transactions
  } = useStore();
  
  const [selectedMonth, setSelectedMonth] = useState(new Date().toISOString().slice(0, 7));
  const [activeTab, setActiveTab] = useState('dashboard');
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [editingTransaction, setEditingTransaction] = useState(null);
  const [isSettingsOpen, setIsSettingsOpen] = useState(false);
  const [isZakatPaidOpen, setIsZakatPaidOpen] = useState(false);
  const [isLoansOpen, setIsLoansOpen] = useState(false);
  const [includeHistorical, setIncludeHistorical] = useState(false);
  const [isSyncing, setIsSyncing] = useState(false);
  const cloudOn = isCloudSyncOn(); // false = local-only mode (no login, no sync)

  // ---- Month-end bank balance reminder ----
  // `now` is refreshed when the app is reopened/refocused and every 30 minutes, so the
  // reminder appears on the last day of the month (or the next time the app is opened).
  const [now, setNow] = useState(() => new Date());
  // Months the user pressed "Remind me later" for: hidden until the app is reopened.
  const [snoozedMonths, setSnoozedMonths] = useState([]);

  // ---- Phone back button (Android app) ----
  // Closes the top-most pop-up first, then returns to the Dashboard tab, then exits.
  // The handler reads the latest screen state through this ref (filled in below).
  const backRef = useRef({});
  useEffect(() => {
    if (!Capacitor.isNativePlatform()) return;
    let listener = null;
    let cancelled = false;
    CapacitorApp.addListener('backButton', () => {
      const b = backRef.current;
      if (b.reminderOpen) b.snoozeReminder();
      else if (b.isModalOpen) b.closeAddModal();
      else if (b.isZakatPaidOpen) b.closeZakatPaid();
      else if (b.isLoansOpen) b.closeLoans();
      else if (b.isSettingsOpen) b.closeSettings();
      else if (b.activeTab !== 'dashboard') b.goDashboard();
      else CapacitorApp.exitApp();
    }).then((l) => { if (cancelled) l.remove(); else listener = l; });
    return () => { cancelled = true; if (listener) listener.remove(); };
  }, []);

  useEffect(() => {
    initSettings().then(() => loadData());
  }, [loadData]);

  useEffect(() => {
    const refresh = () => setNow(new Date());
    const onVisible = () => { if (document.visibilityState === 'visible') refresh(); };
    document.addEventListener('visibilitychange', onVisible);
    const timer = setInterval(refresh, 30 * 60 * 1000);
    return () => {
      document.removeEventListener('visibilitychange', onVisible);
      clearInterval(timer);
    };
  }, []);

  const handleManualSync = async () => {
    setIsSyncing(true);
    try {
      await syncAll();
      await loadData();
    } catch (err) {
      console.error('Manual sync failed:', err);
      alert('Sync problem: ' + (err.message || err));
    } finally {
      // Always switch the spinner off, even if the sync failed.
      setIsSyncing(false);
    }
  };

  if (isLoading) return <LoadingScreen message="Loading Finance ERP..." />;

  const buckets = getBucketBalances(selectedMonth);
  const zakat = calculateZakat(selectedMonth);
  const { totals, grandTotal } = getCategoryTotals(selectedMonth, includeHistorical);
  const monthTransactions = getMonthTransactions(selectedMonth, includeHistorical);
  const totalDebt = getTotalDebtPayable();
  const allCategories = getCategories();

  // The Range report's category filter should also list categories that were removed
  // from the Bucket Map but still exist on past transactions.
  const rangeCategories = [
    ...allCategories,
    ...[...new Set(transactions.filter(t => t.deleted !== true).map(t => t.category).filter(Boolean))]
      .filter(c => !allCategories.includes(c))
      .sort((a, b) => a.localeCompare(b)),
  ];

  // Which month (if any) should we ask a bank balance for right now?
  const reminderEnabled = settings.balance_reminder_enabled !== false; // on unless switched off
  const dueReminderMonths = reminderEnabled
    ? getDueReminderMonths(now, settings.month_balances || {}, settings.balance_reminder_skipped || [])

    : [];
  const reminderMonth = dueReminderMonths[0] || null;

  backRef.current = {

    snoozeReminder: () => setSnoozedMonths(prev => [...prev, reminderMonth]),
    isModalOpen,
    closeAddModal: () => { setIsModalOpen(false); setEditingTransaction(null); },
    isZakatPaidOpen,
    closeZakatPaid: () => setIsZakatPaidOpen(false),
    isLoansOpen,
    closeLoans: () => setIsLoansOpen(false),
    isSettingsOpen,
    closeSettings: () => setIsSettingsOpen(false),
    activeTab,
    goDashboard: () => setActiveTab('dashboard'),
  };

  const handleSaveReminderBalance = async (monthKey, amount) => {
    const latest = useStore.getState().settings.month_balances || {};
    await updateSetting('month_balances', { ...latest, [monthKey]: amount });
  };

  const handleSkipReminderMonth = async (monthKey) => {
    const latest = useStore.getState().settings.balance_reminder_skipped || [];
    if (!latest.includes(monthKey)) {
      await updateSetting('balance_reminder_skipped', [...latest, monthKey]);
    }
  };

  const handleEdit = (t) => {
    setEditingTransaction(t);
    setIsModalOpen(true);
  };

  const handleDelete = async (id) => {
    if (window.confirm('Delete this transaction?')) {
      await deleteTransaction(id);
    }
  };

  // The payment must be dated INSIDE the month being viewed, because that is
  // where the zakat calculation looks for it. For the current month that is
  // right now; for any other month it is that month's last day.
  const zakatPaymentDate = () => {
    const currentMonth = new Date().toISOString().slice(0, 7);
    if (selectedMonth === currentMonth) return new Date().toISOString();
    const [y, m] = selectedMonth.split('-').map(Number);
    return new Date(Date.UTC(y, m, 0, 12, 0, 0)).toISOString();
  };

  const handleMarkZakatPaid = async (amount) => {
    await addTransaction({
      type: 'Expense',
      category: 'Personal',
      subcategory: 'Zakat',
      amount_original: amount,
      currency: 'MVR',
      bucket: 'Financial/Obligatory',
      notes: `Zakat paid for ${selectedMonth}`,
      date: zakatPaymentDate()
    });
    setIsZakatPaidOpen(false);
  };

  return (
    <div className="min-h-screen bg-gray-50 pb-24 font-sans overflow-x-hidden">

          <h1 className="text-xl font-bold text-gray-800">Finance ERP</h1>
          <div className="flex items-center gap-2">
            <div className="flex-1 min-w-0">

            </div>
            {cloudOn && (
              <button 
                onClick={handleManualSync}
                disabled={isSyncing}
                className="p-2 text-gray-500 hover:text-gray-800 hover:bg-gray-100 rounded-lg flex-shrink-0"

              >
                <RefreshCw size={20} className={isSyncing ? "animate-spin" : ""} />

            )}
            <button

              {loans.length > 0 && (
                <span className="absolute -top-0.5 -right-0.5 bg-red-500 text-white text-[10px] rounded-full w-4 h-4 flex items-center justify-center font-bold">
                  {loans.length}
                </span>
              )}
            </button>
            <button 
              onClick={() => setIsSettingsOpen(true)}
              className="p-2 text-gray-500 hover:text-gray-800 hover:bg-gray-100 rounded-lg flex-shrink-0"
            >
              <Settings size={20} />
            </button>
          </div>
          {cloudOn && <SyncStatus />}
        </div>
      </header>

      <AppTabs active={activeTab} onChange={setActiveTab} tabs={TABS} />

      <div className="p-3 max-w-4xl mx-auto overflow-hidden">
        {activeTab === 'dashboard' && (
          <>
            <div className="flex justify-end mb-2">
              <label className="flex items-center gap-2 text-xs text-gray-600 select-none bg-white px-3 py-1.5 rounded-lg shadow-sm">
                <input
                  type="checkbox"
                  checked={includeHistorical}
                  onChange={(e) => setIncludeHistorical(e.target.checked)}
                  className="accent-blue-600"
                />
                Include Historical
              </label>
            </div>

            <ZakatCard
              zakat={zakat}
              selectedMonth={selectedMonth}
              onMarkPaid={() => setIsZakatPaidOpen(true)}
            />
            <TotalDebtCard
              totalDebt={totalDebt}
              loans={loans}
              onManage={() => setIsLoansOpen(true)}
            />
            <CategoryTotals totals={totals} grandTotal={grandTotal} />
            <BucketBalances buckets={buckets} />
            <RecentTransactions
              transactions={monthTransactions}
              onEdit={handleEdit}
              onDelete={handleDelete}
            />
          </>
        )}

        {activeTab === 'insights' && <InsightsTab selectedMonth={selectedMonth} />}
        {activeTab === 'summary' && <YearlySummary />}
        {activeTab === 'range' && <RangeReport allCategories={rangeCategories} />}
      </div>

      <button
        onClick={() => { setEditingTransaction(null); setIsModalOpen(true); }}
        className="fixed bottom-6 right-6 bg-blue-600 text-white p-4 rounded-full shadow-lg hover:bg-blue-700 transition-transform active:scale-95 z-20"
      >
        <Plus size={28} />
      </button>

      {isModalOpen && (
        <AddTransactionModal
          onClose={() => { setIsModalOpen(false); setEditingTransaction(null); }}
          onSave={async (data) => {
            if (editingTransaction) {
              await updateTransaction(editingTransaction.id, data);
            } else {
              await addTransaction(data);
            }
            setIsModalOpen(false);
            setEditingTransaction(null);
          }}
          settings={settings}
          initialData={editingTransaction}
          getBucketFromMap={getBucketFromMap}
          getSubcategoriesFor={getSubcategoriesFor}
          getCategories={getCategories}
          loans={loans}
        />
      )}

      {isSettingsOpen && (
        <SettingsModal
          onClose={() => setIsSettingsOpen(false)}
          settings={settings}
          updateSetting={updateSetting}
          bucketMap={bucketMap}
          updateBucketFor={useStore.getState().updateBucketFor}
          addCategoryEntry={useStore.getState().addCategoryEntry}
          removeCategoryEntry={useStore.getState().removeCategoryEntry}
          removeWholeCategory={useStore.getState().removeWholeCategory}
          onDataChanged={() => useStore.getState().loadData()}
        />
      )}

      {isZakatPaidOpen && (
        <ZakatPaidModal
          onClose={() => setIsZakatPaidOpen(false)}
          onConfirm={handleMarkZakatPaid}
          suggestedAmount={zakat.zakatDue}
        />
      )}

      {reminderMonth && !isSettingsOpen && (
        <BalanceReminderModal
          key={reminderMonth}
          monthKey={reminderMonth}
          remaining={dueReminderMonths.length}
          onSave={handleSaveReminderBalance}
          onLater={() => setSnoozedMonths(prev => [...prev, reminderMonth])}
          onSkip={() => handleSkipReminderMonth(reminderMonth)}
        />
      )}

      {isLoansOpen && (
        <LoansModal
          onClose={() => setIsLoansOpen(false)}
          loans={loans}
          getLoanPaid={getLoanPaid}
          addLoan={useStore.getState().addLoan}
          updateLoan={useStore.getState().updateLoan}
          deleteLoan={useStore.getState().deleteLoan}
        />
      )}
    </div>
  );
}

function App() {
  return (
    <AuthGate>
      <MainApp />
    </AuthGate>
  );
}

export default App;

import { useState } from 'react';
import { X, ArrowDownCircle, ArrowUpCircle } from 'lucide-react';

import { autoLoanUuidFor, findDadLoan, isBappaExpense, BAPPA_LOAN_THRESHOLD } from '../store/useStore';

export function AddTransactionModal({
  onClose,
  onSave,
  settings,
  initialData,
  getBucketFromMap,
  getSubcategoriesFor,
  getCategories,
  loans,
}) {
  const [type, setType] = useState(initialData?.type || 'Expense');
  const [category, setCategory] = useState(initialData?.category || 'Personal');
  const [subcategory, setSubcategory] = useState(initialData?.subcategory || '');
  const [amount, setAmount] = useState(initialData?.amount_original || '');
  const [currency, setCurrency] = useState(initialData?.currency || 'MVR');
  const [notes, setNotes] = useState(initialData?.notes || '');
  const [loanUuid, setLoanUuid] = useState(initialData?.loan_uuid || '');

  const isIncome = type === 'Income';
  const allCategories = getCategories();
  let availableCategories = isIncome
    ? ['Income']
    : allCategories.filter(c => c !== 'Income');

  let availableSubcategories = getSubcategoriesFor(category);

  // Editing an OLD transaction whose category or subcategory has since been removed:
  // keep showing its own values so the form isn't blank and nothing gets changed by accident.
  if (initialData && initialData.type === type) {
    if (initialData.category && !availableCategories.includes(initialData.category)) {
      availableCategories = [...availableCategories, initialData.category];
    }
    if (
      initialData.category === category &&
      initialData.subcategory &&
      !availableSubcategories.includes(initialData.subcategory)
    ) {
      availableSubcategories = [...availableSubcategories, initialData.subcategory];
    }
  }

  // If the category/subcategory is unchanged from what was saved, keep the bucket that
  // was saved with the transaction (removed categories would otherwise fall back to
  // "Discretionary" and silently change old data).
  const isUnchangedEntry =
    !!initialData &&

      subcategory,

      exchange_rate_at_time: initialData?.exchange_rate_at_time,
    });
  };

  const loanDisplay = loans.map(l => ({
    id: l.uuid,
    label: `${l.name} (RF ${l.total_amount.toFixed(2)})`,

      </div>
    </div>
  );
}

import { useState, useEffect, useRef } from 'react';
import { liveQuery } from 'dexie';
import { supabase } from '../lib/supabaseClient';
import { syncAll } from '../lib/sync';
import { db } from '../db/database';
import { useStore } from '../store/useStore';
import { Loader2, Mail, AlertCircle, KeyRound } from 'lucide-react';
import { Capacitor } from '@capacitor/core';
import { App as CapacitorApp } from '@capacitor/app';
import { LoadingScreen } from './LoadingScreen';
import { isCloudSyncOn, HAS_AUTO_LOGIN, AUTO_LOGIN_EMAIL, AUTO_LOGIN_PASSWORD } from '../lib/cloudMode';

// How long to wait after your last change before uploading it.
const PUSH_DELAY_MS = 3000;
// Coming back to the app after at least this long triggers a quiet refresh.
const RETURN_REFRESH_AFTER_MS = 15000;
// While the app is open and visible, check the cloud this often.
const POLL_INTERVAL_MS = 5 * 60 * 1000;

// In the installed Android app the email link must open the APP (not a browser).
// This address is registered with Android (AndroidManifest.xml) and must also be
// added to Supabase -> Authentication -> URL Configuration -> Redirect URLs.
const NATIVE_REDIRECT = 'financeerp://auth-callback';
const IS_NATIVE_APP = Capacitor.isNativePlatform();

const PENDING_TABLES = ['transactions', 'settings', 'loans', 'bucket_map'];

// With cloud sync OFF there is no login at all: the app just opens.
// With it ON, the sign-in + syncing flow below is used (automatic when auto-login
// details are configured, otherwise the login screen).
export function AuthGate({ children }) {
  if (!isCloudSyncOn()) return <>{children}</>;
  return <CloudAuthGate>{children}</CloudAuthGate>;
}

function CloudAuthGate({ children }) {
  const [session, setSession] = useState(null);
  const [email, setEmail] = useState('');
  const [loading, setLoading] = useState(true);
  const [message, setMessage] = useState('');
  const [messageOk, setMessageOk] = useState(false);
  const [codeSent, setCodeSent] = useState(false);
  const [code, setCode] = useState('');
  const [busy, setBusy] = useState(false);
  const [syncError, setSyncError] = useState(null);
  // True when auto-login could not reach the internet: the app then runs on the data
  // already on the phone and tries again when the connection comes back.
  const [offlineNoSession, setOfflineNoSession] = useState(false);

  // Remembers who is signed in, so we can tell "same person, tab came back
  // into view" apart from "someone actually just logged in".
  const userIdRef = useRef(null);

  // blocking = true  -> full-screen spinner (only used the first time the app opens)
  // blocking = false -> sync quietly in the background; the screen never flashes
  const triggerSync = async (blocking) => {
    setSyncError(null);
    if (blocking) setLoading(true);
    try {
      await syncAll();
      if (!blocking) {
        // Background sync finished: refresh the on-screen data without a loading screen.
        await useStore.getState().loadData();
      }
    } catch (e) {
      console.error('Sync failed:', e);
      // Being offline is normal for a phone. Only show the red bar for real errors,
      // or when the person asked for the sync themselves (blocking).
      if (blocking || navigator.onLine) {
        setSyncError(e.message || String(e));
      }
    } finally {
      if (blocking) setLoading(false);
    }
  };

  // Sign in automatically with the email/password from .env.local (if configured).
  const autoLogin = async () => {
    try {
      const { error } = await supabase.auth.signInWithPassword({
        email: AUTO_LOGIN_EMAIL,
        password: AUTO_LOGIN_PASSWORD,
      });
      if (!error) {
        setOfflineNoSession(false);
        return; // SIGNED_IN fires below and starts the first sync
      }
      if (error.name === 'AuthRetryableFetchError' || !navigator.onLine) {
        setOfflineNoSession(true); // no internet: carry on with local data for now
      } else {
        setMessageOk(false);
        setMessage('Automatic sign-in failed: ' + error.message + '. You can sign in with an emailed link or code instead.');
      }
    } catch (err) {
      console.error('Auto-login failed:', err);
      setOfflineNoSession(true);
    }
    setLoading(false);
  };

  useEffect(() => {
    supabase.auth.getSession().then(({ data: { session } }) => {
      setSession(session);
      if (session) {
        userIdRef.current = session.user.id;
        triggerSync(true);
      } else if (HAS_AUTO_LOGIN) {
        autoLogin();
      } else {
        setLoading(false);
      }
    });

    const { data: { subscription } } = supabase.auth.onAuthStateChange((event, newSession) => {
      console.log('Auth Event:', event);

      if (event === 'SIGNED_IN' && newSession) {
        // Coming back to the tab re-fires SIGNED_IN for the same user.
        // That is NOT a new login, so do nothing visible.
        if (userIdRef.current === newSession.user.id) {
          setSession(newSession);
          return;
        }

        // A genuinely new login (e.g. clicking the magic link).
        userIdRef.current = newSession.user.id;
        setSession(newSession);
        triggerSync(true);
      }

      if (event === 'SIGNED_OUT') {
        userIdRef.current = null;
        setSession(null);
      }
    });

    return () => subscription.unsubscribe();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // If we started offline without a session, try the automatic sign-in again as soon
  // as the connection is back or the app is brought back to the screen.
  useEffect(() => {
    if (!HAS_AUTO_LOGIN || !offlineNoSession || session) return;
    const retry = () => {
      if (navigator.onLine && document.visibilityState === 'visible') autoLogin();
    };
    window.addEventListener('online', retry);
    document.addEventListener('visibilitychange', retry);
    return () => {
      window.removeEventListener('online', retry);
      document.removeEventListener('visibilitychange', retry);
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [offlineNoSession, session]);

  // ---------- Finish sign-in when the email link opens the app ----------
  // The link in the email sends you to financeerp://auth-callback#access_token=...
  // Android hands that address to this app, and we use it to sign in.
  const lastHandledUrl = useRef('');

  const handleAuthUrl = async (url) => {
    if (!url || !url.startsWith('financeerp://') || url === lastHandledUrl.current) return;
    lastHandledUrl.current = url;

    const [beforeHash, hashPart = ''] = url.split('#');
    const queryPart = beforeHash.includes('?') ? beforeHash.split('?')[1] : '';
    const params = new URLSearchParams(queryPart);
    new URLSearchParams(hashPart).forEach((v, k) => params.set(k, v));

    try {
      if (params.get('error_description') || params.get('error')) {
        setMessageOk(false);
        setMessage(
          (params.get('error_description') || params.get('error')).replace(/\+/g, ' ') +
          ' - request a new link and try again.'
        );
        return;
      }
      if (params.get('access_token') && params.get('refresh_token')) {
        const { error } = await supabase.auth.setSession({
          access_token: params.get('access_token'),
          refresh_token: params.get('refresh_token'),
        });
        if (error) throw error;
      } else if (params.get('code')) {
        const { error } = await supabase.auth.exchangeCodeForSession(params.get('code'));
        if (error) throw error;
      }
      // On success Supabase fires SIGNED_IN and the code above takes over.
    } catch (err) {
      console.error('Deep link sign-in failed:', err);
      setMessageOk(false);
      setMessage('Could not sign in from the link: ' + (err.message || err) + '. Use the code from the email instead.');
    }
  };

  useEffect(() => {
    if (!IS_NATIVE_APP) return;
    let listener = null;
    let cancelled = false;

    CapacitorApp.addListener('appUrlOpen', (event) => handleAuthUrl(event.url)).then((l) => {
      if (cancelled) l.remove(); else listener = l;
    });
    // If the app was closed and the link launched it, pick up that address too.
    CapacitorApp.getLaunchUrl().then((res) => handleAuthUrl(res?.url)).catch(() => {});

    return () => {
      cancelled = true;
      if (listener) listener.remove();
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // ---------- Automatic syncing (only while signed in) ----------
  const signedIn = !!session;

  useEffect(() => {
    if (!signedIn) return;

    let pushTimer = null;
    let hiddenAt = null;

    const quietSync = () => {
      if (!navigator.onLine) return;
      triggerSync(false);
    };

    // 1. Upload shortly after you save/edit anything.
    //    liveQuery re-runs whenever the local database changes. We count rows
    //    still marked "pending" (not yet uploaded). If any exist, wait a few
    //    seconds (so several quick edits go up together) and then sync.
    const subscription = liveQuery(async () => {
      let total = 0;
      for (const t of PENDING_TABLES) {
        total += await db.table(t).where('syncStatus').equals('pending').count();
      }
      return total;
    }).subscribe({
      next: (pendingCount) => {
        clearTimeout(pushTimer);
        if (pendingCount > 0) {
          pushTimer = setTimeout(quietSync, PUSH_DELAY_MS);
        }
      },
      error: (err) => console.error('Pending watcher error:', err),
    });

    // 2. Refresh when you come back to the app after a while.
    const onVisibility = () => {
      if (document.visibilityState === 'hidden') {
        hiddenAt = Date.now();
      } else if (hiddenAt && Date.now() - hiddenAt > RETURN_REFRESH_AFTER_MS) {
        hiddenAt = null;
        quietSync();
      }
    };
    document.addEventListener('visibilitychange', onVisibility);

    // 3. Sync when the internet comes back.
    window.addEventListener('online', quietSync);

    // 4. Check the cloud every few minutes while the app is open and visible.
    const poll = setInterval(() => {
      if (document.visibilityState === 'visible') quietSync();
    }, POLL_INTERVAL_MS);

    return () => {
      clearTimeout(pushTimer);
      subscription.unsubscribe();
      document.removeEventListener('visibilitychange', onVisibility);
      window.removeEventListener('online', quietSync);
      clearInterval(poll);
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [signedIn]);

  const handleLogin = async (e) => {
    e.preventDefault();
    setMessage('');
    setBusy(true);
    const { error } = await supabase.auth.signInWithOtp({
      email: email.trim(),
      options: {
        // App: the link opens the installed app. Website: it returns to the website.
        emailRedirectTo: IS_NATIVE_APP ? NATIVE_REDIRECT : window.location.origin,
      },
    });
    setBusy(false);
    if (error) {
      setMessageOk(false);
      setMessage(error.message);
    } else {
      setMessageOk(true);
      setCodeSent(true);
      setMessage('Check your email. Tap the link, or type the code from the email below.');
    }
  };

  // Sign in by typing the code from the email (works in any browser or app).
  const handleVerifyCode = async (e) => {
    e.preventDefault();
    setMessage('');
    setBusy(true);
    const { error } = await supabase.auth.verifyOtp({
      email: email.trim(),
      token: code.trim(),
      type: 'email',
    });
    setBusy(false);
    if (error) {
      setMessageOk(false);
      setMessage(error.message);
    }
    // On success Supabase fires SIGNED_IN and the app continues by itself.
  };

  if (loading) {
    return (
      <LoadingScreen
        message={session ? 'Syncing your data from the cloud...' : 'Loading...'}
      />
    );
  }

  if (!session && !offlineNoSession) {
    return (
      <div className="min-h-screen flex flex-col items-center justify-center bg-gray-50 p-4">
        <div className="bg-white p-6 rounded-2xl shadow-md w-full max-w-sm">
          <h1 className="text-2xl font-bold text-center mb-2 text-gray-800">Finance ERP</h1>
          <p className="text-center text-gray-500 text-sm mb-6">Sign in to sync your data.</p>

          <form onSubmit={handleLogin} className="space-y-4">
            <div>
              <label className="text-xs font-semibold text-gray-500 mb-1 block">Email Address</label>
              <div className="relative">
                <Mail
                  size={16}
                  className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400"
                />
                <input
                  type="email"
                  required
                  value={email}
                  onChange={(e) => setEmail(e.target.value)}
                  className="w-full pl-9 p-3 bg-gray-50 rounded-lg border border-gray-200 outline-none focus:border-blue-500"
                  placeholder="you@example.com"
                />
              </div>
            </div>

            <button
              type="submit"
              className="w-full bg-blue-600 text-white py-3 rounded-lg font-bold hover:bg-blue-700 transition"
            >
              {busy && !codeSent ? 'Sending...' : codeSent ? 'Send a new link / code' : 'Send login link'}
            </button>
          </form>

          {codeSent && (
            <form onSubmit={handleVerifyCode} className="space-y-3 mt-5 pt-5 border-t border-gray-100">
              <label className="text-xs font-semibold text-gray-500 block">Code from the email</label>
              <div className="relative">
                <KeyRound
                  size={16}
                  className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400"
                />
                <input
                  type="text"
                  inputMode="numeric"
                  autoComplete="one-time-code"
                  required
                  value={code}
                  onChange={(e) => setCode(e.target.value.replace(/\D/g, ''))}
                  className="w-full pl-9 p-3 bg-gray-50 rounded-lg border border-gray-200 outline-none focus:border-blue-500 tracking-widest text-lg"
                  placeholder="123456"
                />
              </div>
              <button
                type="submit"
                disabled={busy || code.length < 6}
                className="w-full bg-gray-800 text-white py-3 rounded-lg font-bold hover:bg-gray-900 transition disabled:opacity-40"
              >
                {busy ? 'Checking...' : 'Sign in with code'}
              </button>
            </form>
          )}

          {message && (
            <div
              className={`mt-4 p-3 rounded-lg text-sm text-center ${
                messageOk
                  ? 'bg-green-50 text-green-700'
                  : 'bg-red-50 text-red-700'
              }`}
            >
              {message}
            </div>
          )}
        </div>
      </div>
    );
  }

  return (
    <>
      {syncError && (
        <div className="fixed top-0 left-0 right-0 z-[100] bg-red-600 text-white text-sm p-3 flex items-center justify-between gap-2">
          <div className="flex items-center gap-2 min-w-0">
            <AlertCircle size={16} className="flex-shrink-0" />
            <span className="truncate">Sync failed: {syncError}. Showing local data.</span>
          </div>
          <button
            onClick={() => triggerSync(false)}
            className="bg-white/20 hover:bg-white/30 px-3 py-1 rounded text-xs font-semibold flex-shrink-0"
          >
            Retry
          </button>
        </div>
      )}
      {children}
    </>
  );
}

  const [saving, setSaving] = useState(false);

  const handleSave = async (e) => {

import { useState, useRef, useEffect } from 'react';
import { ChevronDown, Search } from 'lucide-react';
import { useClickOutside } from '../hooks/useClickOutside';

export function CustomSelect({ 
  value, 
  onChange, 
  options, 
  placeholder = '-- Select --',
  required = false,
  className = '',
  buttonClassName = '',
  searchable = true,
  disabled = false
}) {
  const [isOpen, setIsOpen] = useState(false);
  const [search, setSearch] = useState('');
  const ref = useRef(null);
  const searchRef = useRef(null);
  useClickOutside(ref, () => setIsOpen(false));

  // Reset search when closed
  useEffect(() => {
    if (!isOpen) setSearch('');
  }, [isOpen]);

  // Focus the search box automatically on desktop only. On a phone, auto-focusing
  // opens the on-screen keyboard, which squashes the screen and shifts the list while
  // you are scrolling or tapping - that made taps on an option get lost. On a phone,
  // tap the search box yourself when you want the keyboard.
  useEffect(() => {
    if (!isOpen || !searchable) return;
    const isTouchDevice =
      typeof window !== 'undefined' &&
      window.matchMedia &&
      window.matchMedia('(pointer: coarse)').matches;
    if (isTouchDevice) return;
    const t = setTimeout(() => searchRef.current?.focus(), 50);
    return () => clearTimeout(t);
  }, [isOpen, searchable]);

  // Always show search if searchable is true
  const showSearch = searchable;
  
  // Remove blanks and duplicates (duplicate entries confuse React's list handling
  // and can make filtering/selection act strangely), keeping the original order.
  const cleanOptions = [...new Set((options || []).map(o => String(o ?? '')).filter(o => o !== ''))];

  const query = search.trim().toLowerCase();
  const filtered = showSearch && query
    ? cleanOptions.filter(o => o.toLowerCase().includes(query))
    : cleanOptions;

  const displayValue = value || placeholder;
  const isPlaceholder = !value;

  const handleSelect = (opt) => {
    onChange(opt);
    setIsOpen(false);
  };

  return (
    <div ref={ref} className={`relative w-full ${className}`}>
      <button
        type="button"
        onClick={() => !disabled && setIsOpen(!isOpen)}
        disabled={disabled}
        className={`w-full p-3 bg-gray-50 rounded-lg border border-gray-200 text-left flex justify-between items-center gap-2 ${
          disabled ? 'opacity-50 cursor-not-allowed' : 'hover:border-gray-300'
        } ${buttonClassName}`}
      >
        <span className={`text-sm truncate ${isPlaceholder ? 'text-gray-400' : 'text-gray-700'}`}>
          {displayValue}
        </span>
        <ChevronDown 
          size={16} 
          className={`text-gray-500 flex-shrink-0 transition-transform ${isOpen ? 'rotate-180' : ''}`} 
        />
      </button>

      {isOpen && (
        <div className="absolute z-50 mt-1 w-full bg-white rounded-lg shadow-lg border border-gray-200 overflow-hidden">
          {showSearch && (
            <div className="relative border-b border-gray-100">
              <Search size={14} className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400" />
              <input
                ref={searchRef}
                type="text"
                value={search}
                onChange={(e) => setSearch(e.target.value)}
                onKeyDown={(e) => {
                  // Enter inside the search box must not submit the surrounding form.
                  if (e.key === 'Enter') {
                    e.preventDefault();
                    if (filtered.length === 1) handleSelect(filtered[0]);
                  }
                }}
                placeholder="Search..."
                autoComplete="off"
                autoCorrect="off"
                spellCheck={false}
                className="w-full pl-8 pr-3 py-2 text-base sm:text-sm outline-none"
              />
            </div>
          )}

          {/* Pressing an option must not steal focus from the search box, otherwise the
              on-screen keyboard collapses and shifts the list before the tap registers. */}
          <div
            className="max-h-60 overflow-y-auto overscroll-contain"
            style={{ WebkitOverflowScrolling: 'touch' }}
            onMouseDown={(e) => e.preventDefault()}
          >
            {!required && (
              <button
                type="button"
                onClick={() => handleSelect('')}
                className={`w-full text-left px-3 py-2 text-sm hover:bg-gray-50 ${
                  !value ? 'bg-blue-50 text-blue-700 font-semibold' : 'text-gray-400 italic'
                }`}
              >
                {placeholder}
              </button>
            )}

            {filtered.length === 0 ? (
              <div className="px-3 py-4 text-xs text-gray-400 text-center">No matches.</div>
            ) : (
              filtered.map((opt, i) => (
                <button
                  key={`${i}-${opt}`}
                  type="button"
                  onClick={() => handleSelect(opt)}
                  className={`w-full text-left px-3 py-3 sm:py-2 text-sm hover:bg-gray-50 truncate ${
                    opt === value ? 'bg-blue-50 text-blue-700 font-semibold' : 'text-gray-700'
                  }`}
                >
                  {opt}
                </button>
              ))
            )}
          </div>
        </div>
      )}
    </div>
  );
}

import { useState, useRef, useEffect } from 'react';
import { ChevronDown, ChevronLeft, ChevronRight } from 'lucide-react';
import { useClickOutside } from '../hooks/useClickOutside';

const MONTH_NAMES_LONG = [
  'January', 'February', 'March', 'April', 'May', 'June',
  'July', 'August', 'September', 'October', 'November', 'December'
];
const MONTH_NAMES_SHORT = [
  'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
  'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'
];

export function DatePicker({ value, onChange }) {
  const [isOpen, setIsOpen] = useState(false);
  const [view, setView] = useState('days'); // 'days' | 'months' | 'years'
  
  // The currently selected date (for highlighting)
  const d = value ? new Date(value) : new Date();
  const validDate = isNaN(d.getTime()) ? new Date() : d;

  const [viewMonth, setViewMonth] = useState({
    year: validDate.getFullYear(),
    month: validDate.getMonth(),
  });

  const ref = useRef(null);
  useClickOutside(ref, () => setIsOpen(false));

  // Sync viewMonth when the value changes externally
  useEffect(() => {
    if (value) {
      const parsed = new Date(value);
      if (!isNaN(parsed)) {
        setViewMonth({ year: parsed.getFullYear(), month: parsed.getMonth() });
      }
    }
  }, [value]);

  // Reset the view to days when opening
  useEffect(() => {
    if (isOpen) setView('days');
  }, [isOpen]);

  const displayValue = value
    ? validDate.toLocaleDateString(undefined, { year: 'numeric', month: 'short', day: 'numeric' })
    : 'Select date';

  // --- Calendar Grid Calculation ---
  const daysInMonth = new Date(viewMonth.year, viewMonth.month + 1, 0).getDate();
  const firstDayOfWeek = new Date(viewMonth.year, viewMonth.month, 1).getDay();
  const days = [];
  for (let i = 0; i < firstDayOfWeek; i++) days.push(null);
  for (let d = 1; d <= daysInMonth; d++) days.push(d);

  const handleSelectDay = (day) => {
    const m = String(viewMonth.month + 1).padStart(2, '0');
    const dd = String(day).padStart(2, '0');
    onChange(`${viewMonth.year}-${m}-${dd}`);
    setIsOpen(false);
  };

  const goPrevMonth = () => {
    setViewMonth(v => v.month === 0 ? { year: v.year - 1, month: 11 } : { year: v.year, month: v.month - 1 });
  };

  const goNextMonth = () => {
    setViewMonth(v => v.month === 11 ? { year: v.year + 1, month: 0 } : { year: v.year, month: v.month + 1 });
  };

  const todayStr = new Date().toISOString().slice(0, 10);

  // --- Year Grid Calculation ---
  // Show a block of 12 years (e.g. 2015 - 2026)
  const yearBlockStart = Math.floor(viewMonth.year / 12) * 12;
  const years = Array.from({ length: 12 }, (_, i) => yearBlockStart + i);

  return (
    <div ref={ref} className="relative w-full">
      <button
        type="button"
        onClick={() => setIsOpen(!isOpen)}
        className="w-full p-3 bg-gray-50 rounded-lg border border-gray-200 text-left flex justify-between items-center hover:border-gray-300 transition-colors"
      >
        <span className={`text-sm truncate ${!value ? 'text-gray-400' : 'text-gray-700'}`}>
          {displayValue}
        </span>
        <ChevronDown size={16} className={`text-gray-500 flex-shrink-0 transition-transform ${isOpen ? 'rotate-180' : ''}`} />
      </button>

      {isOpen && (
        <div className="absolute z-50 mt-1 w-[280px] bg-white rounded-xl shadow-xl border border-gray-200 p-3">
          
          {/* Header */}
          <div className="flex justify-between items-center mb-3">
            {view === 'days' ? (
              <button type="button" onClick={goPrevMonth} className="p-1.5 text-gray-500 hover:bg-gray-100 rounded-lg w-8 h-8 flex items-center justify-center">
                <ChevronLeft size={16} />
              </button>
            ) : (
              <div className="w-8" /> // Spacer
            )}

            <button
              type="button"
              onClick={() => setView(view === 'days' ? 'months' : 'years')}
              className="font-bold text-sm text-gray-800 hover:bg-gray-100 px-2 py-1 rounded transition-colors"
            >
              {view === 'days' ? `${MONTH_NAMES_LONG[viewMonth.month]} ${viewMonth.year}` : 
               view === 'months' ? `${viewMonth.year}` : 
               `${yearBlockStart} - ${yearBlockStart + 11}`}
            </button>

            {view === 'days' ? (
              <button type="button" onClick={goNextMonth} className="p-1.5 text-gray-500 hover:bg-gray-100 rounded-lg w-8 h-8 flex items-center justify-center">
                <ChevronRight size={16} />
              </button>
            ) : (
              <div className="w-8" /> // Spacer
            )}
          </div>

          {/* Views */}
          {view === 'days' && (
            <>
              <div className="grid grid-cols-7 gap-1 mb-1">
                {['S','M','T','W','T','F','S'].map((d, i) => (
                  <div key={i} className="text-[10px] text-gray-400 text-center font-semibold">{d}</div>
                ))}
              </div>

              <div className="grid grid-cols-7 gap-1">
                {days.map((day, idx) => {
                  if (day === null) return <div key={idx}></div>;
                  const m = String(viewMonth.month + 1).padStart(2, '0');
                  const dd = String(day).padStart(2, '0');
                  const dateStr = `${viewMonth.year}-${m}-${dd}`;
                  const isSelected = dateStr === (value || '');
                  const isToday = dateStr === todayStr;
                  return (
                    <button
                      key={idx}
                      type="button"
                      onClick={() => handleSelectDay(day)}
                      className={`aspect-square text-xs rounded-lg flex items-center justify-center transition-colors ${
                        isSelected
                          ? 'bg-blue-600 text-white font-bold shadow-md'
                          : isToday
                            ? 'bg-blue-50 text-blue-600 font-semibold border border-blue-100'
                            : 'hover:bg-gray-100 text-gray-700'
                      }`}
                    >
                      {day}
                    </button>
                  );
                })}
              </div>
            </>
          )}

          {view === 'months' && (
            <div className="grid grid-cols-3 gap-2 mt-2">
              {MONTH_NAMES_SHORT.map((month, idx) => {
                const isSelected = viewMonth.month === idx;

  const [preview, setPreview] = useState(null);
  const [skipDuplicates, setSkipDuplicates] = useState(true);

  const DB_FIELDS = [
    { key: 'date', label: 'Date', required: true },
    { key: 'category', label: 'Category', required: true },
    { key: 'subcategory', label: 'Subcategory', required: false },
    { key: 'amount_original', label: 'Amount', required: true },
    { key: 'currency', label: 'Currency', required: false },
    { key: 'notes', label: 'Notes', required: false },

      const data = new Uint8Array(event.target.result);
      const workbook = XLSX.read(data, { type: 'array' });
      setSheets(workbook.SheetNames);
      setSelectedSheet(workbook.SheetNames[0]);

      if (workbook.SheetNames.length === 1) {

      } else {

      }

      if (idx === -1) idx = headerRow.findIndex(h => h.toLowerCase() === targetLabel);

          const lower = h.toLowerCase();

    if (!selectedLabel || selectedLabel === '-- Skip --') {
      setMapping({ ...mapping, [dbField]: undefined });
      return;
    }
    const idx = columnOptions.indexOf(selectedLabel) - 1;
    setMapping({ ...mapping, [dbField]: idx });
  };

  // Reads every spreadsheet row and turns it into a clean transaction.
  // Rows with missing/invalid data are counted as "invalid" and skipped.
  const buildBatch = () => {
    let invalid = 0;
    const batch = [];

    for (let i = 0; i < rows.length; i++) {
      const row = rows[i];
      if (!row || row.length === 0) continue;

      const dateVal = mapping.date !== undefined ? row[mapping.date] : null;
      const catVal = mapping.category !== undefined ? row[mapping.category] : null;
      const subVal = mapping.subcategory !== undefined ? row[mapping.subcategory] : '';
      const amtVal = mapping.amount_original !== undefined ? row[mapping.amount_original] : null;
      const currVal = mapping.currency !== undefined ? row[mapping.currency] : 'MVR';
      const notesVal = mapping.notes !== undefined ? row[mapping.notes] : '';

      if (!dateVal || !catVal || amtVal === null || amtVal === '') {
        invalid++;
        continue;
      }

      let finalDate;
      if (typeof dateVal === 'number') {
        const excelEpoch = new Date(Date.UTC(1899, 11, 30));
        const msPerDay = 24 * 60 * 60 * 1000;
        finalDate = new Date(excelEpoch.getTime() + dateVal * msPerDay).toISOString();
      } else if (typeof dateVal === 'string') {

      const currency = String(currVal).toUpperCase() === 'USD' ? 'USD' : 'MVR';
      // For historical rows, we set exchange_rate to 15.42 but the rate is stored.
      // The app's logic will use the stored amount_mvr.
      const rate = currency === 'USD' ? 15.42 : 1;

        }

      });

    }

      }

      setImportResult({
        success: successCount,
        errors: preview.invalid,
        skippedDuplicates: skipDuplicates ? preview.duplicates : 0,
      });
      setStep(4);
    } catch (err) {
      console.error('[Import] FATAL:', err);
      setImportResult({ success: successCount, errors: preview.invalid + 1, fatalError: err.message || String(err) });
      setStep(4);
    } finally {
      setIsImporting(false);

    }

const MONTH_NAMES = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

// The 2018-2022 rows are whole-year lumps (notes like "2018 aggregates") parked on
// one date. They can't be split into months, so monthly insights skip them.
// They still count in Yearly Summary.
function isYearlyLump(t) {
  return typeof t.notes === 'string' && /aggregates?\s*$/i.test(t.notes.trim());
}

const amt = (t) => Number(t.amount_mvr) || 0;

function shiftMonth(monthString, delta) {
  const [y, m] = monthString.split('-').map(Number);
  const d = new Date(y, m - 1 + delta, 1);
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
}

function monthLabel(monthString) {
  const [y, m] = monthString.split('-').map(Number);
  return `${MONTH_NAMES[m - 1]} ${y}`;
}

function shortMonthLabel(monthString) {
  const [, m] = monthString.split('-').map(Number);
  return MONTH_NAMES[m - 1];
}

function StatRow({ label, current, previous, tone }) {
  const delta = current - previous;
  const pct = previous !== 0 ? (delta / Math.abs(previous)) * 100 : null;
  const isUp = delta > 0;
  const isFlat = Math.abs(delta) < 0.005;

  const toneColor = tone === 'income' ? 'text-green-600' : tone === 'expense' ? 'text-red-600' : 'text-gray-800';

  let DeltaIcon = Minus;
  let deltaColor = 'text-gray-400';
  if (!isFlat) {
    if (tone === 'income') {
      DeltaIcon = isUp ? TrendingUp : TrendingDown;
      deltaColor = isUp ? 'text-green-600' : 'text-red-500';
    } else if (tone === 'expense') {
      DeltaIcon = isUp ? TrendingUp : TrendingDown;
      deltaColor = isUp ? 'text-red-500' : 'text-green-600';
    } else {
      DeltaIcon = isUp ? TrendingUp : TrendingDown;
      deltaColor = isUp ? 'text-green-600' : 'text-red-500';
    }
  }

  return (
    <div className="flex justify-between items-center py-2 border-b border-gray-100 last:border-0">
      <span className="text-sm text-gray-600 font-semibold">{label}</span>
      <div className="flex items-center gap-3">
        <span className={`text-sm font-bold ${toneColor}`}>{formatMoney(current)}</span>
        <div className={`flex items-center gap-0.5 text-xs font-semibold ${deltaColor} min-w-[72px] justify-end`}>
          <DeltaIcon size={12} />
          {isFlat ? (
            <span>—</span>
          ) : (
            <span>
              {pct !== null ? `${pct >= 0 ? '+' : ''}${pct.toFixed(0)}%` : formatMoney(delta)}
            </span>
          )}
        </div>
      </div>
    </div>
  );
}

      Discretionary: rows.filter(t => t.type === 'Expense' && t.bucket === 'Discretionary').reduce((s, t) => s + amt(t), 0),
      Financial: rows.filter(t => t.type === 'Expense' && t.bucket === 'Financial/Obligatory').reduce((s, t) => s + amt(t), 0),
      Savings: 0,
    };
    const build = (name) => {
      const a = alloc[name];
      const sp = spent[name];
      const pct = a > 0 ? (sp / a) * 100 : 0;
      return { name, allocated: a, spent: sp, pct, over: sp > a && a > 0 };
    };
    return [build('Essential'), build('Discretionary'), build('Financial'), build('Savings')];
  }, [budgetTx, thisMonth, settings]);

  const sourceCounts = useMemo(() => {
    const rows = monthlyTx.filter(t => t.date.startsWith(thisMonth));
    const historical = rows.filter(t => t.is_historical).length;
    return { historical, live: rows.length - historical };
  }, [monthlyTx, thisMonth]);

  // Is the selected period one of the yearly-lump years (no monthly detail exists)?
  const onlyYearlyLumps = useMemo(
    () => transactions.some(
      t => t.deleted !== true && isYearlyLump(t) &&
        (t.date.startsWith(thisMonth.slice(0, 4)))
    ),
    [transactions, thisMonth]
  );

  const hasAnyData = monthAgg.this.count > 0 || monthAgg.last.count > 0;

  if (!hasAnyData) {
    return (
      <div className="text-center p-10 text-gray-400 bg-white rounded-xl shadow-sm text-sm">
        {onlyYearlyLumps
          ? `${thisMonth.slice(0, 4)} was recorded as yearly totals only, so there is no month-by-month detail to analyse. See the Yearly Summary instead.`
          : `No transaction data for ${monthLabel(thisMonth)} or ${monthLabel(lastMonth)}. Add transactions or switch months to see insights.`}
      </div>
    );
  }

  return (
    <div className="space-y-1">
      <SectionCard title={`Month vs Month`} icon={Calendar} tone="info">
        <div className="text-xs text-gray-400 mb-2">
          {monthLabel(lastMonth)} → {monthLabel(thisMonth)}
        </div>
        <StatRow
          label="Income"
          current={monthAgg.this.income}
          previous={monthAgg.last.income}
          tone="income"
        />
        <StatRow
          label="Expenses"
          current={monthAgg.this.expense}
          previous={monthAgg.last.expense}
          tone="expense"
        />
        <StatRow
          label="Net"
          current={monthAgg.this.net}
          previous={monthAgg.last.net}
          tone="net"
        />
        <div className="text-[11px] text-gray-400 mt-2">
          This month: {monthAgg.this.count} txn • Last month: {monthAgg.last.count} txn
        </div>
        {sourceCounts.historical > 0 && (
          <div className="text-[11px] text-gray-400 mt-0.5">
            This month includes {sourceCounts.historical} imported historical and {sourceCounts.live} live entries.
          </div>
        )}
      </SectionCard>

      <SectionCard title={`Top Subcategories — ${monthLabel(thisMonth)}`} icon={Flame} tone="warning">
        {topSubs.rows.length === 0 ? (

            {topSubs.rows.map((r, i) => {

              return (

                    className="absolute top-0 left-0 h-full bg-orange-50 rounded-md"
                    style={{ width: `${pct}%` }}
                  ></div>
                  <div className="relative flex justify-between items-center gap-2 py-1.5">
                    <div className="flex items-center gap-2 min-w-0">
                      <span className="text-[10px] font-bold text-gray-400 w-4 flex-shrink-0">
                        #{i + 1}
                      </span>
                      <div className="min-w-0">
                        <div className="text-sm font-semibold text-gray-700 truncate">
                          {r.subcategory}
                        </div>
                        <div className="text-[10px] text-gray-400 truncate">
                          {r.category} • {r.count} txn

const MONTH_NAMES = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

      <button 
        type="button"
        onClick={() => setIsOpen(!isOpen)}

      >
        <span>
          {MONTH_NAMES[parseInt(selectedMonth.slice(5, 7)) - 1]} {selectedMonth.slice(0, 4)}
        </span>
        <ChevronDown size={16} className={`text-gray-500 transition-transform ${isOpen ? 'rotate-180' : ''}`} />
      </button>
      
      {isOpen && (
        <div className="absolute z-50 mt-1 w-full bg-white rounded-lg shadow-lg border border-gray-200 p-3">
          <div className="flex justify-between items-center mb-3">
            <button 
              type="button" 
              onClick={() => setPickerYear(y => y - 1)} 
              className="p-1 text-gray-500 hover:bg-gray-100 rounded w-8 h-8 flex items-center justify-center"
            >
              ←
            </button>
            <span className="font-bold text-sm">{pickerYear}</span>

              type="button" 
              onClick={() => setPickerYear(y => y + 1)} 
              className="p-1 text-gray-500 hover:bg-gray-100 rounded w-8 h-8 flex items-center justify-center"
            >
              →
            </button>
          </div>
          <div className="grid grid-cols-3 gap-1">
            {MONTH_NAMES.map((m, i) => {
              const monthVal = `${pickerYear}-${String(i + 1).padStart(2, '0')}`;
              const isSelected = selectedMonth === monthVal;
              return (
                <button
                  key={i}
                  type="button"
                  onClick={() => { onChange(monthVal); setIsOpen(false); }}
                  className={`p-2 text-sm rounded ${
                    isSelected 
                      ? 'bg-blue-500 text-white' 
                      : 'hover:bg-gray-100 text-gray-700'
                  }`}
                >
                  {m}
                </button>
              );
            })}
          </div>
        </div>
      )}
    </div>
  );
}

import { useState, useEffect } from 'react';
import { liveQuery } from 'dexie';
import { db } from '../db/database';
import { getLastOkSync, SYNC_EVENT } from '../lib/sync';

const TABLES = ['transactions', 'settings', 'loans', 'bucket_map'];

function timeAgo(iso) {
  if (!iso) return 'not yet';
  const seconds = Math.max(0, Math.round((Date.now() - new Date(iso).getTime()) / 1000));
  if (seconds < 45) return 'just now';
  const minutes = Math.round(seconds / 60);
  if (minutes < 60) return `${minutes} min ago`;
  const hours = Math.round(minutes / 60);
  if (hours < 24) return `${hours} hr ago`;
  return `${Math.round(hours / 24)} days ago`;
}

/**
 * A small status line: "Synced 2 min ago" or "3 changes waiting to upload".
 * It only displays information; it never starts a sync itself.
 */
export function SyncStatus() {
  const [pending, setPending] = useState(0);
  const [lastOk, setLastOk] = useState(getLastOkSync());
  const [lastError, setLastError] = useState(null);
  const [online, setOnline] = useState(navigator.onLine);
  const [, setTick] = useState(0);

  useEffect(() => {
    const sub = liveQuery(async () => {
      let total = 0;
      for (const t of TABLES) {
        total += await db.table(t).where('syncStatus').equals('pending').count();
      }
      return total;
    }).subscribe({ next: setPending, error: () => {} });

    const onSync = (e) => {
      setLastOk(getLastOkSync());
      setLastError(e.detail && !e.detail.ok ? e.detail.error : null);
    };
    const onOnline = () => setOnline(true);
    const onOffline = () => setOnline(false);

    window.addEventListener(SYNC_EVENT, onSync);
    window.addEventListener('online', onOnline);
    window.addEventListener('offline', onOffline);
    // Refresh the "x min ago" wording every 30 seconds.
    const timer = setInterval(() => setTick(n => n + 1), 30000);

    return () => {
      sub.unsubscribe();
      window.removeEventListener(SYNC_EVENT, onSync);
      window.removeEventListener('online', onOnline);
      window.removeEventListener('offline', onOffline);
      clearInterval(timer);
    };
  }, []);

  let text;
  let color = 'text-gray-400';

  if (!online) {
    text = pending > 0
      ? `Offline • ${pending} change${pending === 1 ? '' : 's'} saved on this device, will upload when you're back online`
      : 'Offline • changes will upload when you reconnect';
    color = 'text-orange-500';
  } else if (lastError) {
    text = `Last sync had a problem: ${lastError}`;
    color = 'text-red-500';
  } else if (pending > 0) {
    text = `${pending} change${pending === 1 ? '' : 's'} waiting to upload • last synced ${timeAgo(lastOk)}`;
    color = 'text-orange-500';
  } else {
    text = `Synced ${timeAgo(lastOk)}`;
    color = 'text-green-600';
  }

  return <div className={`text-[11px] truncate ${color}`}>{text}</div>;
}

// OPTIONAL HARDENING (versions 203 + 204): make duplicate uuids impossible.
//
// Version 203 first removes any rows that share a uuid (keeping the newest).
// Version 204 then adds the "unique" rule. They are two separate steps on purpose:
// the browser refuses to add a unique rule while duplicates still exist, so the
// clean-up has to happen first.
// ---------------------------------------------------------------------------
db.version(203).stores({}).upgrade(async tx => {
  for (const tableName of ['transactions', 'loans', 'bucket_map']) {
    const rows = await tx.table(tableName).toArray();
    const keep = new Map();
    const remove = [];

    for (const row of rows) {
      if (!row.uuid) continue; // rows without a uuid are ignored by a unique index
      const prev = keep.get(row.uuid);
      if (!prev) {
        keep.set(row.uuid, row);
        continue;
      }
      const prevTime = new Date(prev.updated_at || 0).getTime();
      const thisTime = new Date(row.updated_at || 0).getTime();
      if (thisTime > prevTime) {
        remove.push(prev.id);
        keep.set(row.uuid, row);
      } else {
        remove.push(row.id);
      }
    }

    if (remove.length > 0) {
      await tx.table(tableName).bulkDelete(remove);
    }
  }
});

import { useEffect } from 'react';

export function useClickOutside(ref, onClose) {
  useEffect(() => {
    const handler = (e) => {
      if (ref.current && !ref.current.contains(e.target)) onClose();
    };
    document.addEventListener('mousedown', handler);
    document.addEventListener('touchstart', handler);
    return () => {
      document.removeEventListener('mousedown', handler);
      document.removeEventListener('touchstart', handler);
    };
  }, [ref, onClose]);
}

// Cloud sync / login switch.
//
// AUTO-LOGIN: if the file .env.local (in the project folder, never uploaded to GitHub)
// contains VITE_AUTO_LOGIN_EMAIL and VITE_AUTO_LOGIN_PASSWORD, the app signs in by
// itself and syncs, with no login screen. Those two values are baked into the app when
// you build it, so anyone who gets your APK file could extract them.
//
// The Settings switch can still turn cloud sync on or off on a device:
//   - nothing chosen yet -> ON if auto-login is configured, otherwise OFF
//   - OFF: no login, data stays on this device only
//   - ON : sign in (automatically, or with the login screen) and sync with Supabase
// The choice is stored on THIS device only, and changing it reloads the app.

const KEY = 'finance_erp_cloud_sync';

export const AUTO_LOGIN_EMAIL = (import.meta.env.VITE_AUTO_LOGIN_EMAIL || '').trim();
export const AUTO_LOGIN_PASSWORD = import.meta.env.VITE_AUTO_LOGIN_PASSWORD || '';
export const HAS_AUTO_LOGIN = !!(AUTO_LOGIN_EMAIL && AUTO_LOGIN_PASSWORD);

export function isCloudSyncOn() {
  try {
    const saved = localStorage.getItem(KEY);
    if (saved === 'on') return true;
    if (saved === 'off') return false;
    return HAS_AUTO_LOGIN; // no choice made yet
  } catch {
    return HAS_AUTO_LOGIN;
  }
}

export function setCloudSync(on) {
  try {
    localStorage.setItem(KEY, on ? 'on' : 'off');
  } catch (err) {
    console.error('Could not save cloud sync choice:', err);
  }
  window.location.reload();
}

async function fetchAllRows(table, userId, sinceISO) {
  const allRows = [];
  let from = 0;

  while (true) {
    let query = supabase
      .from(table)
      .select('*')
      .eq('user_id', userId)
      // order by id as a tie-breaker so paging is stable
      .order('updated_at', { ascending: true })
      .order('id', { ascending: true })
      .range(from, from + PAGE_SIZE - 1);

    if (sinceISO) {
      query = query.gt('updated_at', sinceISO);
    }

    const { data, error } = await query;
    if (error) throw error;

    if (!data || data.length === 0) break;
    allRows.push(...data);

    if (data.length < PAGE_SIZE) break;
    from += PAGE_SIZE;
  }

  return allRows;
}

/**
 * Local self-repair. Runs before every sync.
 *  1. Rows that somehow have no uuid get one (e.g. imported rows).
 *  2. If two local rows share the same uuid, keep the newest, delete the rest.
 * This quietly cleans up duplicates created by the old buggy pull.
 */
async function cleanupLocal() {
  for (const table of UUID_TABLES) {
    const rows = await db.table(table).toArray();
    const seen = new Map();
    const toDelete = [];
    const needUuid = [];

    for (const r of rows) {
      if (!r.uuid) {
        needUuid.push(r.id);
        continue;
      }
      const prev = seen.get(r.uuid);
      if (!prev) {
        seen.set(r.uuid, r);
        continue;
      }
      const prevTime = new Date(prev.updated_at || 0).getTime();
      const thisTime = new Date(r.updated_at || 0).getTime();
      // Keep whichever is newer. On a tie, keep the one we saw first.
      if (thisTime > prevTime) {
        toDelete.push(prev.id);
        seen.set(r.uuid, r);
      } else {
        toDelete.push(r.id);
      }
    }

    if (toDelete.length > 0) {
      console.warn(`[sync] Removing ${toDelete.length} duplicate local rows from ${table}`);
      await db.table(table).bulkDelete(toDelete);
    }

    if (needUuid.length > 0) {
      const nowISO = new Date().toISOString();
      await db.table(table).where('id').anyOf(needUuid).modify(row => {
        row.uuid = uuidv4();
        row.syncStatus = 'pending';
        row.updated_at = nowISO;
        if (row.deleted === undefined) row.deleted = false;
      });
    }
  }
}

/**
 * liveOnly = true uploads ONLY things you typed in yourself (live transactions
 * and loans). Used as a safety step before a device replaces its local data.
 */
export async function pushChanges({ liveOnly = false } = {}) {
  const { data: { session } } = await supabase.auth.getSession();
  if (!session) return false;

  const now = new Date().toISOString();
  let allSuccessful = true;

  for (const table of TABLES) {
    let pending = await db.table(table).where('syncStatus').equals('pending').toArray();
    if (liveOnly) {
      if (table === 'transactions') pending = pending.filter(r => !r.is_historical);
      else if (table !== 'loans') continue;
    }
    if (pending.length === 0) continue;

    const toPush = pending.map(row => {
      // eslint-disable-next-line no-unused-vars
      const { id, syncStatus, uuid, ...rest } = row;
      // Settings saved by older code may have no uuid; derive a stable one from the key.
      const cloudId = uuid || (table === 'settings' ? stableUuid('setting|' + row.key) : uuidv4());
      return {
        id: cloudId,
        ...rest,
        user_id: session.user.id,
        updated_at: now,
      };
    });

    for (let i = 0; i < toPush.length; i += PUSH_CHUNK) {
      const chunk = toPush.slice(i, i + PUSH_CHUNK);
      const chunkPending = pending.slice(i, i + PUSH_CHUNK);

      try {
        if (table === 'settings') {
          const { error } = await supabase
            .from(table)
            .upsert(chunk, { onConflict: 'user_id,key' });
          if (error) throw error;
          const keys = chunkPending.map(p => p.key);
          await db.settings.where('key').anyOf(keys).modify({ syncStatus: 'synced' });
        } else {
          const { error } = await supabase.from(table).upsert(chunk);
          if (error) throw error;
          const ids = chunkPending.map(p => p.id);
          await db.table(table).where('id').anyOf(ids).modify({ syncStatus: 'synced' });
        }
      } catch (err) {
        console.error(`Error pushing ${table} chunk starting at ${i}:`, err.message);
        allSuccessful = false;
      }
    }
  }

  return allSuccessful;
}

export async function pullChanges(isFirstSync = false) {
  const { data: { session } } = await supabase.auth.getSession();
  if (!session) return;

  const lastSync = getLastSync();
  const pullStartedAt = new Date().toISOString();

  let sinceISO = null;
  if (!isFirstSync && lastSync) {
    sinceISO = new Date(new Date(lastSync).getTime() - PULL_OVERLAP_MS).toISOString();
  }

  // Step 1: download everything first. If any download fails we stop BEFORE
  // touching the local database, so a network hiccup can never leave you half-wiped.
  const fetched = {};
  for (const table of TABLES) {
    fetched[table] = await fetchAllRows(table, session.user.id, sinceISO);
  }

  // Safety stop: never replace this device's data with a cloud answer that
  // looks wrong (empty, or far smaller than what this device already holds).
  if (isFirstSync) {
    const localSynced = await db.transactions.where('syncStatus').equals('synced').count();
    const cloudCount = fetched.transactions.length;
    if (cloudCount === 0 || (localSynced > 100 && cloudCount < localSynced * 0.5)) {
      throw new Error(
        `Safety stop: the cloud returned ${cloudCount} transactions but this device has ${localSynced}. ` +
        'Your local data was NOT replaced. Download a backup in Settings, then try again.'
      );
    }
  }

  // Step 2: write into the local database, one table at a time, all-or-nothing.
  for (const table of TABLES) {
    const rows = fetched[table];

    await db.transaction('rw', db.table(table), async () => {
      if (isFirstSync) {
        await db.table(table).clear();
      }
      if (!rows || rows.length === 0) return;

      // Look up what we already have locally, so we can UPDATE instead of INSERT.
      // (This lookup is the fix. The old code skipped it, so every pull inserted
      // a brand-new copy of rows that already existed.)
      const existing = await db.table(table).toArray();
      const localByKey = new Map();
      for (const r of existing) {
        localByKey.set(table === 'settings' ? r.key : r.uuid, r);
      }

      const toPut = [];
      for (const row of rows) {
        // eslint-disable-next-line no-unused-vars
        const { user_id, id: cloudId, ...rest } = row;
        const local = localByKey.get(table === 'settings' ? row.key : cloudId);

        // Don't overwrite an unsynced local edit that is newer than the cloud copy.
        if (local && local.syncStatus === 'pending') {
          const localTime = new Date(local.updated_at || 0).getTime();
          const cloudTime = new Date(row.updated_at || 0).getTime();
          if (localTime > cloudTime) continue;
        }

        const out = { ...rest, uuid: cloudId, syncStatus: 'synced' };
        // Reusing the local row's id is what makes put() replace it.
        if (local && table !== 'settings') out.id = local.id;
        toPut.push(out);
      }

      if (toPut.length > 0) {
        await db.table(table).bulkPut(toPut);
      }
    });
  }

  setLastSync(pullStartedAt);
}

// Returns true (all good), false (finished but some changes didn't upload), or null (not signed in).
async function runSync() {
  const { data: { session } } = await supabase.auth.getSession();
  if (!session) return null;

  // One-time reset after this fix is deployed: forget the old sync marker
  // so this device does one clean full download.
  if (localStorage.getItem(EPOCH_KEY) !== SYNC_EPOCH) {
    localStorage.removeItem(LAST_SYNC_KEY);
  }

  await cleanupLocal();

  const lastSync = getLastSync();

  if (!lastSync) {
    // First sync on this device (or a forced clean re-download).
    const { count, error } = await supabase
      .from('transactions')
      .select('*', { count: 'exact', head: true })
      .eq('user_id', session.user.id);

    if (error) throw error;

    if (count === 0) {
      // Cloud is empty: this device's data becomes the cloud's data.
      await initSettings();
      for (const t of TABLES) {
        await db.table(t).toCollection().modify({ syncStatus: 'pending' });
      }
      const ok = await pushChanges();
      if (!ok) throw new Error('Could not upload your data to the cloud.');
      setLastSync(new Date().toISOString());
    } else {
      // Cloud has data: wipe local and take the cloud's copy.
      // Safety first: upload anything you typed in that hasn't reached the cloud yet,
      // otherwise the wipe below would delete it. If this upload fails, stop.
      const savedLive = await pushChanges({ liveOnly: true });
      if (!savedLive) {
        throw new Error(
          'This device has entries that could not be uploaded, so it was not reset. Check your connection and try again.'
        );
      }
      await pullChanges(true);
      // Make sure historical seed rows exist (same ids on every device),
      // then upload anything new.
      await initSettings();
      await pushChanges();
    }

    localStorage.setItem(EPOCH_KEY, SYNC_EPOCH);
    return true;
  }

  // Normal sync: push first. If the push fails, do NOT pull.
  const pushOk = await pushChanges();
  if (pushOk) {
    await pullChanges(false);
  } else {
    console.warn('Push failed. Skipping pull so local edits are not overwritten.');
  }
  return pushOk;
}

let runningSync = null;

/**
 * Only ONE sync may run at a time. If sync is requested while another is
 * running (the app asks twice on startup, and again when you switch tabs),
 * the second request just waits for the first one.
 * navigator.locks extends this protection across browser tabs.
 */
export function syncAll() {
  if (runningSync) return runningSync;

  const task = () => runSync();
  const wrapped = (navigator.locks && navigator.locks.request)
    ? navigator.locks.request('finance-erp-sync', task)
    : task();

  runningSync = Promise.resolve(wrapped)
    .then((result) => {
      if (result === true) announceSync(true);
      else if (result === false) announceSync(false, 'Some changes could not be uploaded. Will retry.');
      return result;
    })
    .catch((err) => {
      announceSync(false, err.message || String(err));
      throw err;
    })
    .finally(() => { runningSync = null; });
  return runningSync;
}

// Category entries the app's own logic depends on. They can be re-bucketed but not removed:
//  - the whole "Income" category (the Income tab of Add Transaction needs it)
//  - the whole "Personal" category (the default category)
//  - Personal > Zakat       (used by the zakat card and "Mark Paid")
//  - Personal > Loan Repayment (links repayments to loans)
export function isProtectedCategory(category) {
  return category === 'Income' || category === 'Personal';
}
export function isProtectedEntry(category, subcategory) {
  if (category === 'Personal' && (subcategory === 'Zakat' || subcategory === 'Loan Repayment')) return true;
  return false;
}

// The year your saved "Previous Year Monthly Balances" grid belongs to.
// (Entered in Oct 2026 as last year's month-end balances, so 2025.)
const PREV_BALANCES_YEAR = 2025;

// ---- Automatic loan deduction for Family > Bappa ----
// An expense under Family > Bappa ABOVE this amount (in MVR) is counted as a payment
// towards the loan named "Dad". The transaction itself is the payment: it is simply
// linked to that loan, so no second "Loan Repayment" transaction is created.
// Anything at or below this amount is an ordinary expense and is not linked to a loan.
export const BAPPA_LOAN_THRESHOLD = 1100;
const BAPPA_LOAN_NAME = 'dad';

// True if this transaction is the kind that may be deducted from the Dad loan.
export function isBappaExpense(t) {
  return !!t && t.type === 'Expense' && t.category === 'Family' && t.subcategory === 'Bappa';
}

// The loan named "Dad" (ignores capitals and extra spaces; ignores deleted loans).
export function findDadLoan(loans) {
  return (loans || []).find(
    l => l.deleted !== true && String(l.name || '').trim().toLowerCase() === BAPPA_LOAN_NAME
  );
}

// Which loan (uuid) should this transaction be linked to? Returns null if none.
// `amountMvr` is the transaction amount already converted to MVR.
export function autoLoanUuidFor(t, amountMvr, loans) {
  if (!isBappaExpense(t)) return null;
  if (!(Number(amountMvr) > BAPPA_LOAN_THRESHOLD)) return null;
  return findDadLoan(loans)?.uuid || null;
}

    };

    // Re-check the Family > Bappa loan rule whenever such a transaction is saved, so
    // editing the amount (or moving it in/out of Bappa) keeps the Dad loan correct.
    // Other transactions (e.g. normal Loan Repayments) are left exactly as they were.
    if (isBappaExpense(updated)) {
      updated.loan_uuid = autoLoanUuidFor(updated, amount_mvr, get().loans);
    }

    await db.transactions.put(updated);

    set((state) => ({
      transactions: state.transactions.map(t => (t.id === id ? updated : t)),
    }));
  },

  deleteTransaction: async (id) => {
    await db.transactions.update(id, {
      deleted: true,
      syncStatus: 'pending',
      updated_at: new Date().toISOString(),
    });
    set((state) => ({
      transactions: state.transactions.filter(t => t.id !== id),
    }));
  },

  // ============ CALCULATIONS ============
  getBucketBalances: (monthString) => {
    const { transactions, settings } = get();
    const monthTransactions = transactions.filter(
      t => t.date.startsWith(monthString) && !t.is_historical && t.deleted !== true
    );

    const totalIncome = monthTransactions
      .filter(t => t.type === 'Income')
      .reduce((sum, t) => sum + (t.amount_mvr || 0), 0);

    const ePct = Number(settings.essential_pct) || 40;
    const dPct = Number(settings.discretionary_pct) || 10;
    const fPct = Number(settings.financial_pct) || 15;
    const sPct = Number(settings.savings_pct) || 35;

    const essentialAllocated = totalIncome * (ePct / 100);
    const discretionaryAllocated = totalIncome * (dPct / 100);
    const financialAllocated = totalIncome * (fPct / 100);
    const savingsAllocated = totalIncome * (sPct / 100);

    const essentialSpent = monthTransactions
      .filter(t => t.type === 'Expense' && t.bucket === 'Essential')
      .reduce((sum, t) => sum + (t.amount_mvr || 0), 0);
    const discretionarySpent = monthTransactions
      .filter(t => t.type === 'Expense' && t.bucket === 'Discretionary')
      .reduce((sum, t) => sum + (t.amount_mvr || 0), 0);
    const financialSpent = monthTransactions
      .filter(t => t.type === 'Expense' && t.bucket === 'Financial/Obligatory')
      .reduce((sum, t) => sum + (t.amount_mvr || 0), 0);

    return {
      Essential: { allocated: essentialAllocated, spent: essentialSpent, remaining: essentialAllocated - essentialSpent },
      Discretionary: { allocated: discretionaryAllocated, spent: discretionarySpent, remaining: discretionaryAllocated - discretionarySpent },
      Financial: { allocated: financialAllocated, spent: financialSpent, remaining: financialAllocated - financialSpent },
      Savings: { allocated: savingsAllocated, spent: 0, remaining: savingsAllocated },
    };
  },

  getCategoryTotals: (monthString, includeHistorical = false) => {
    const { transactions } = get();
    const monthExpenses = transactions.filter(
      t =>
        t.date.startsWith(monthString) &&
        t.type === 'Expense' &&
        t.deleted !== true &&
        (includeHistorical || !t.is_historical) &&
        Number.isFinite(t.amount_mvr)
    );

    const byCategory = {};

    monthExpenses.forEach(t => {
      const cat = t.category || '(Uncategorized)';
      const sub = t.subcategory || '(No subcategory)';

      if (!byCategory[cat]) {
        byCategory[cat] = { category: cat, amount: 0, subcategories: {} };
      }
      if (!byCategory[cat].subcategories[sub]) {
        byCategory[cat].subcategories[sub] = { subcategory: sub, amount: 0, count: 0 };
      }

      byCategory[cat].amount += t.amount_mvr;
      byCategory[cat].subcategories[sub].amount += t.amount_mvr;
      byCategory[cat].subcategories[sub].count += 1;
    });

    const totals = Object.values(byCategory)
      .map(c => ({
        category: c.category,
        amount: c.amount,
        subcategories: Object.values(c.subcategories).sort((a, b) => b.amount - a.amount),
      }))
      .sort((a, b) => b.amount - a.amount);

    const grandTotal = totals.reduce((sum, item) => sum + item.amount, 0);

    return { totals, grandTotal };
  },

  getMonthTransactions: (monthString, includeHistorical = false) => {
    const { transactions } = get();
    return transactions
      .filter(
        t =>
          t.date.startsWith(monthString) &&
          t.deleted !== true &&
          (includeHistorical || !t.is_historical)
      )
      .sort((a, b) => new Date(b.date) - new Date(a.date));
  },

  // ============ ZAKAT LOGIC ============
  calculateZakat: (monthString) => {
    const { transactions, settings } = get();
    const [yearStr, monthKey] = monthString.split('-');
    const year = Number(yearStr);

    const zakatPaidThisMonth = transactions
      .filter(
        t =>
          t.type === 'Expense' &&
          t.category === 'Personal' &&
          t.subcategory === 'Zakat' &&
          t.date.startsWith(monthString) &&
          !t.is_historical &&
          t.deleted !== true
      )
      .reduce((sum, t) => sum + (t.amount_mvr || 0), 0);

    // Turns a stored value into a number, or null if nothing valid was entered.
    const readBalance = (v) => {
      if (v === undefined || v === null || v === '') return null;
      const n = parseFloat(v);
      return Number.isNaN(n) ? null : n;
    };

    // Bank balance for THIS month, typed in by hand (like cell B15 in the Excel sheet).
    const monthBalances = settings.month_balances || {};
    const enteredBalance = readBalance(monthBalances[monthString]);
    const noBalance = enteredBalance === null;
    const bankBalance = noBalance ? 0 : enteredBalance;

    const debtPayable = get().getTotalDebtPayable();
    const debtReceivable = parseFloat(settings.debt_receivable) || 0;
    const nisab = parseFloat(settings.nisab_threshold) || 10000;

    // Same month last year: prefer a balance entered for that exact year,
    // otherwise use the saved "previous year" grid (only valid for its own year).
    const prevYearBalances = settings.prev_year_balances || {};
    const lastYearEntered = readBalance(monthBalances[`${year - 1}-${monthKey}`]);
    const lastYearSaved =
      year - 1 === PREV_BALANCES_YEAR ? readBalance(prevYearBalances[monthKey]) : null;
    const lastYearBalance = lastYearEntered ?? lastYearSaved ?? 0;

    // Matches the Excel rule: eligible if balance > last year's, or balance > nisab.
    const eligible = !noBalance && (bankBalance > lastYearBalance || bankBalance > nisab);

    const zakatableAmount = bankBalance - debtPayable + debtReceivable;
    const baseZakat = eligible && zakatableAmount > 0 ? zakatableAmount * 0.025 : 0;

    const zakatDue = Math.max(0, baseZakat - zakatPaidThisMonth);
    const fullyPaid = baseZakat > 0 && zakatDue === 0;

    return {
      bankBalance,
      bankBalanceBeforeZakat: bankBalance,
      noBalance,
      debtPayable,
      debtReceivable,
      nisab,
      lastYearBalance,
      zakatableAmount,
      baseZakat,
      zakatPaidThisMonth,
      zakatDue,
      eligible,
      fullyPaid,
    };
  },

  // ============ RANGE + YEARLY ============
  getTransactionsInRange: (fromISO, toISO, opts = {}) => {
    const { includeHistorical = true, type = 'All', category = null } = opts;
    const { transactions } = get();

    let from = null;
    if (fromISO) {
      const d = new Date(fromISO);
      d.setHours(0, 0, 0, 0);
      from = d.getTime();
    }

    let to = null;
    if (toISO) {
      const d = new Date(toISO);
      d.setHours(23, 59, 59, 999);
      to = d.getTime();
    }

    return transactions
      .filter(t => {
        if (t.deleted === true) return false;
        if (!includeHistorical && t.is_historical) return false;
        if (type !== 'All' && t.type !== type) return false;
        if (category && t.category !== category) return false;

        const d = new Date(t.date).getTime();
        if (from !== null && d < from) return false;
        if (to !== null && d > to) return false;
        return true;
      })
      .sort((a, b) => new Date(b.date) - new Date(a.date));
  },

  aggregateRange: (rows) => {
    const buildGroup = (typeFilter) => {
      const map = {};

      rows.forEach(t => {
        if (t.type !== typeFilter) return;
        if (!Number.isFinite(t.amount_mvr)) return;

        const cat = t.category || '(Uncategorized)';
        const sub = t.subcategory || '(No subcategory)';

        if (!map[cat]) map[cat] = { category: cat, total: 0, subcategories: {} };
        if (!map[cat].subcategories[sub]) {
          map[cat].subcategories[sub] = { subcategory: sub, total: 0, count: 0 };
        }

        map[cat].total += t.amount_mvr;
        map[cat].subcategories[sub].total += t.amount_mvr;
        map[cat].subcategories[sub].count += 1;
      });

      const categories = Object.values(map)
        .map(c => ({
          category: c.category,
          total: c.total,
          subcategories: Object.values(c.subcategories).sort((a, b) => b.total - a.total),
        }))
        .sort((a, b) => b.total - a.total);

      const grandTotal = categories.reduce((sum, c) => sum + c.total, 0);
      return { categories, grandTotal };
    };

    return {
      expenses: buildGroup('Expense'),
      incomes: buildGroup('Income'),
    };
  },

  getYearlySummary: (type = 'Expense') => {
    const { transactions } = get();

    const txFiltered = transactions.filter(
      t => t.type === type && t.deleted !== true && Number.isFinite(t.amount_mvr)
    );

    const yearSet = new Set();
    txFiltered.forEach(t => {
      const y = new Date(t.date).getFullYear();
      if (!isNaN(y)) yearSet.add(y);
    });
    const years = Array.from(yearSet).sort((a, b) => a - b);

    const catMap = {};
    txFiltered.forEach(t => {
      const y = new Date(t.date).getFullYear();
      if (isNaN(y)) return;

      const cat = t.category || '(Uncategorized)';
      if (!catMap[cat]) {
        catMap[cat] = { category: cat, byYear: {} };
        years.forEach(yy => {
          catMap[cat].byYear[yy] = 0;
        });
      }
      catMap[cat].byYear[y] = (catMap[cat].byYear[y] || 0) + t.amount_mvr;
    });

    const rows = Object.values(catMap)
      .map(r => {
        const total = years.reduce((sum, y) => sum + (r.byYear[y] || 0), 0);
        const activeYears = years.filter(y => (r.byYear[y] || 0) > 0);
        const avg = activeYears.length > 0 ? total / activeYears.length : 0;
        return { ...r, total, avg };
      })
      .sort((a, b) => b.total - a.total);

    const grandTotalByYear = {};
    years.forEach(y => {
      grandTotalByYear[y] = rows.reduce((sum, r) => sum + (r.byYear[y] || 0), 0);
    });

    const grandTotal = rows.reduce((sum, r) => sum + r.total, 0);

    return { years, rows, grandTotalByYear, grandTotal };
  },

  getSubcategoryYearlySummary: (targetCategory, years, type = 'Expense') => {
    const { transactions } = get();

    const catTx = transactions.filter(
      t =>
        t.type === type &&
        t.category === targetCategory &&
        t.deleted !== true &&
        Number.isFinite(t.amount_mvr)
    );

    const subMap = {};
    catTx.forEach(t => {
      const y = new Date(t.date).getFullYear();
      if (isNaN(y)) return;

      const sub = t.subcategory || '(No subcategory)';
      if (!subMap[sub]) {
        subMap[sub] = { subcategory: sub, byYear: {} };
        years.forEach(yy => {
          subMap[sub].byYear[yy] = 0;
        });
      }
      subMap[sub].byYear[y] = (subMap[sub].byYear[y] || 0) + t.amount_mvr;
    });

    return Object.values(subMap)
      .map(r => {
        const total = years.reduce((sum, y) => sum + (r.byYear[y] || 0), 0);
        const activeYears = years.filter(y => (r.byYear[y] || 0) > 0);
        const avg = activeYears.length > 0 ? total / activeYears.length : 0;
        return { ...r, total, avg };
      })
      .sort((a, b) => b.total - a.total);
  },
}));

/**
 * Saves a copy of ALL your data (transactions, settings, loans, bucket map)
 * into one .json file and downloads it to this device.
 * Returns how many rows of each kind were saved.
 */
export async function downloadBackup() {
  const [transactions, settings, loans, bucket_map] = await Promise.all([
    db.transactions.toArray(),
    db.settings.toArray(),
    db.loans.toArray(),
    db.bucket_map.toArray(),
  ]);

  const backup = {
    app: 'finance-erp',
    format_version: 1,
    exported_at: new Date().toISOString(),
    counts: {
      transactions: transactions.length,
      settings: settings.length,
      loans: loans.length,
      bucket_map: bucket_map.length,
    },
    tables: { transactions, settings, loans, bucket_map },
  };

  const blob = new Blob([JSON.stringify(backup)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const link = document.createElement('a');
  const stamp = new Date().toISOString().slice(0, 10);
  link.href = url;
  link.download = `finance-erp-backup-${stamp}.json`;
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  setTimeout(() => URL.revokeObjectURL(url), 5000);

  return backup.counts;
}

// Month-end "enter your bank balance" reminder (used for the zakat calculation).
//
// A month becomes "due" once its LAST DAY has arrived. If the app isn't opened that
// day, the month stays due and the reminder simply appears the next time the app is
// opened. A month stops being due when a balance has been entered for it, or when
// you press "Skip this month".

// How many past months to look back. Older gaps are not nagged about; they can still
// be filled in from Settings -> Zakat Inputs.
export const REMINDER_LOOKBACK_MONTHS = 3;

const pad = (n) => String(n).padStart(2, '0');

// "YYYY-MM" key for a year and a 0-based month index (handles month going below 0).
function monthKey(year, monthIndex) {
  const d = new Date(year, monthIndex, 1);
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}`;
}

// Months (oldest first, as "YYYY-MM") that still need a bank balance.
//   now       - a Date (uses the device's local calendar day, not UTC)
//   balances  - the saved month_balances object
//   skipped   - array of "YYYY-MM" the user chose to skip
export function getDueReminderMonths(now, balances = {}, skipped = []) {
  const y = now.getFullYear();
  const m = now.getMonth(); // 0-based
  const lastDayOfThisMonth = new Date(y, m + 1, 0).getDate();

  const candidates = [];
  for (let back = REMINDER_LOOKBACK_MONTHS; back >= 1; back--) {
    candidates.push(monthKey(y, m - back)); // earlier months: their last day has passed
  }
  if (now.getDate() === lastDayOfThisMonth) {
    candidates.push(monthKey(y, m)); // this month, but only on its last day
  }

  const hasBalance = (key) => {
    const v = balances ? balances[key] : undefined;
    return v !== '' && v !== null && v !== undefined && !Number.isNaN(Number(v));
  };

  return candidates.filter(key => !hasBalance(key) && !(skipped || []).includes(key));
}

export function monthLabel(key) {
  const [y, mo] = key.split('-').map(Number);
  return new Date(y, mo - 1, 1).toLocaleDateString(undefined, { month: 'long', year: 'numeric' });
}

import * as XLSX from 'xlsx';

export function exportMasterSheet(expenseData, incomeData) {
  const wb = XLSX.utils.book_new();

  const addSheetsForData = (data, prefix) => {
    const { years, rows, grandTotalByYear, grandTotal } = data;
    if (years.length === 0) return; // Skip if no data for this type

    // ---- Sheet 1: Master Matrix ----
    const header = ['Category', ...years, 'Avg', 'Total'];
    const body = rows.map(r => [
      r.category,
      ...years.map(y => r.byYear[y] || 0),
      Number(r.avg.toFixed(2)),
      Number(r.total.toFixed(2)),
    ]);

    const totalRow = [
      `All ${prefix}s`,
      ...years.map(y => grandTotalByYear[y] || 0),
      '',
      Number(grandTotal.toFixed(2)),
    ];

    const master = XLSX.utils.aoa_to_sheet([header, ...body, totalRow]);

    master['!cols'] = [
      { wch: 20 },
      ...years.map(() => ({ wch: 12 })),
      { wch: 12 },
      { wch: 14 },
    ];

    XLSX.utils.book_append_sheet(wb, master, `${prefix} Master`);

    // ---- Sheet 2: Averages & Totals ----
    const summaryHeader = ['Category', 'All Time Avg', 'All Time Total'];
    const summaryBody = rows.map(r => [
      r.category,
      Number(r.avg.toFixed(2)),
      Number(r.total.toFixed(2)),
    ]);
    const summarySheet = XLSX.utils.aoa_to_sheet([summaryHeader, ...summaryBody]);
    summarySheet['!cols'] = [{ wch: 20 }, { wch: 14 }, { wch: 14 }];
    XLSX.utils.book_append_sheet(wb, summarySheet, `${prefix} Averages`);

    // ---- Sheet 3: Year Totals ----
    const yearHeader = ['Year', 'Grand Total'];
    const yearBody = years.map(y => [y, Number((grandTotalByYear[y] || 0).toFixed(2))]);
    const yearSheet = XLSX.utils.aoa_to_sheet([yearHeader, ...yearBody]);
    yearSheet['!cols'] = [{ wch: 10 }, { wch: 16 }];
    XLSX.utils.book_append_sheet(wb, yearSheet, `${prefix} Year Totals`);
  };

  // Generate the sheets for both types
  addSheetsForData(expenseData, 'Expense');
  addSheetsForData(incomeData, 'Income');

  // Save the file
  const stamp = new Date().toISOString().slice(0, 10);
  XLSX.writeFile(wb, `finance-master-${stamp}.xlsx`);
}
